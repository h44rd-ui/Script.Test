local Character

local State = {
    Character = nil,
    Humanoid = nil,
    HRP = nil
}

local function UpdateCharacter()
    Character = Player.Character

    if not Character then
        Humanoid = nil
        RootPart = nil

        State.Character = nil
        State.Humanoid = nil
        State.HRP = nil

        return
    end

    Humanoid =
        Character:FindFirstChildOfClass("Humanoid")

    RootPart =
        Character:FindFirstChild("HumanoidRootPart")

    State.Character = Character
    State.Humanoid = Humanoid
    State.HRP = RootPart
end

    task.wait(0.5)
    UpdateCharacter()
end)

local function GetValidBall(Object)
    if not Object then
        return nil
    end

    if Object:IsA("BasePart") then
        return Object
    end

    if Object:IsA("Model") then
        if Object.PrimaryPart then
            return Object.PrimaryPart
        end

        return Object:FindFirstChildWhichIsA(
            "BasePart",
            true
        )
    end

    return nil
end

local function FindTPS()
    local Folder =
        Workspace:FindFirstChild(
            "WorkspaceLeaderboards"
        )

    if Folder then
        local TPS =
            Folder:FindFirstChild("TPS")

        if TPS then
            return TPS
        end
    end

    for _, Object in ipairs(
        Workspace:GetDescendants()
    ) do
        if Object.Name == "TPS" then
            return Object
        end
    end

    return nil
end

local function GetBall()
    return GetValidBall(
        FindTPS()
    )
end

local ballList = {}

local function scanBalls()
    table.clear(ballList)

    local Folder =
        Workspace:FindFirstChild(
            "WorkspaceLeaderboards"
        )

    if Folder then
        local TPS =
            Folder:FindFirstChild("TPS")

        local Ball =
            GetValidBall(TPS)

        if Ball then
            ballList[#ballList + 1] =
                Ball
        end
    end

    if #ballList == 0 then
        for _, Object in ipairs(
            Workspace:GetDescendants()
        ) do

            if Object.Name == "TPS" then
                local Ball =
                    GetValidBall(Object)

                if Ball then
                    local Exists = false

                    for i = 1, #ballList do
                        if ballList[i] == Ball then
                            Exists = true
                            break
                        end
                    end

                    if not Exists then
                        ballList[#ballList + 1] =
                            Ball
                    end
                end
            end
        end
    end
end

local GKBotoes = {}

local function EscanearBotoesGK()
    GKBotoes = {}

    local PlayerGui =
        Player:FindFirstChildOfClass(
            "PlayerGui"
        )

    if not PlayerGui then
        return
    end

    for _, Object in ipairs(
        PlayerGui:GetDescendants()
    ) do

        if Object:IsA("TextButton")
            or Object:IsA("ImageButton") then

            local Name =
                string.lower(Object.Name)

            local Text = ""

            pcall(function()
                Text =
                    string.lower(
                        Object.Text or ""
                    )
            end)

            if Name:find("gk")
                or Name:find("c2")
                or Text:find("dive")
                or Text:find("catch")
                or Text:find("high")
                or Text:find("low")
                or Text:find("reflex")
                or Text:find("forward")
                or Text:find("front")
                or Text:find("rush") then

                table.insert(
                    GKBotoes,
                    {
                        Button = Object,
                        Nome = Object.Name,
                        Texto = Text
                    }
                )
            end
        end
    end
end

local function EncontrarBotaoPorTexto(Texto)
    local Procurado =
        string.lower(Texto)

    for _, Info in ipairs(GKBotoes) do
        if Info.Button
            and Info.Button.Parent
            and Info.Texto == Procurado then

            return Info.Button
        end
    end

    for _, Info in ipairs(GKBotoes) do
        if Info.Button
            and Info.Button.Parent
            and Info.Texto
            and Info.Texto:find(
                Procurado,
                1,
                true
            ) then

            return Info.Button
        end
    end

    return nil
end

local function GetDivePrediction(Ball)
    if not Ball or not RootPart then
        return nil
    end

    local Position =
        Ball.Position

    local Velocity =
        Ball.AssemblyLinearVelocity

    local Speed =
        Velocity.Magnitude

    if Speed <
        Dive.MinBallSpeed then

        return nil
    end

    local Distance =
        (Position - RootPart.Position)
        .Magnitude

    local TravelTime =
        math.clamp(
            Distance / Speed,
            0,
            Dive.Prediction
        )

    local Predicted =
        Position +
        Velocity * TravelTime

    local Relative =
        Predicted -
        RootPart.Position

    local Horizontal =
        Vector3.new(
            Relative.X,
            0,
            Relative.Z
        )

    local Right =
        RootPart.CFrame.RightVector

    local RightHorizontal =
        Vector3.new(
            Right.X,
            0,
            Right.Z
        )

    local Side =
        Horizontal:Dot(
            RightHorizontal
        )

    return {
        Position = Predicted,
        Side = Side,
        Height = Relative.Y,
        IsCentral =
            math.abs(Side)
            <= Dive.CenterWidth,
        IsHigh =
            Relative.Y >=
            Dive.HighHeight,
        IsFront =
            Relative.Y <=
            Dive.FrontHeight
    }
end

local function GetDiveAction(Data)
    if not Data then
        return nil
    end

    if Data.IsHigh
        and Data.IsCentral then

        return DiveButtons.HighCatch
    end

    if Data.IsFront
        and Data.IsCentral then

        return DiveButtons.Front
    end

    if Data.IsHigh then
        return Data.Side < 0
            and DiveButtons.HighLeft
            or DiveButtons.HighRight
    end

    return Data.Side < 0
        and DiveButtons.LowLeft
        or DiveButtons.LowRight
end

local function ExecuteDiveAction(Action)
    if not Action then
        return false
    end

    local Button =
        EncontrarBotaoPorTexto(Action)

    if not Button then
        EscanearBotoesGK()
        Button =
            EncontrarBotaoPorTexto(Action)
    end

    if not Button then
        return false
    end

    return ClicarBotao(Button)
end

local function IsBallComingToGoalkeeper(Ball)
    if not Ball or not RootPart then
        return false
    end

    local Velocity =
        Vector3.new(
            Ball.AssemblyLinearVelocity.X,
            0,
            Ball.AssemblyLinearVelocity.Z
        )

    if Velocity.Magnitude <
        Dive.MinBallSpeed then

        return false
    end

    local Direction =
        RootPart.Position -
        Ball.Position

    Direction =
        Vector3.new(
            Direction.X,
            0,
            Direction.Z
        )

    if Direction.Magnitude <= 0 then
        return false
    end

    local Dot =
        Velocity.Unit:Dot(
            Direction.Unit
        )

    return Dot >=
        Dive.MinApproachDot
end

local function ExecuteAutoDive(Ball)
    local Data =
        GetDivePrediction(Ball)

    if not Data then
        return
    end

    local Action =
        GetDiveAction(Data)

    if ExecuteDiveAction(Action) then
        Dive.LastAction =
            tick()
    end
end

local AutoDiveConnection

local function StartAutoDive()
    if AutoDiveConnection then
        return
    end

    AutoDiveConnection =
        RunService.Heartbeat:Connect(
            function()

                if not Dive.Enabled then
                    return
                end

                if not RootPart
                    or not RootPart.Parent then

                    UpdateCharacter()
                    return
                end

                if tick() -
                    Dive.LastAction <
                    Dive.Cooldown then

                    return
                end

                local Ball =
                    GetBall()

                if not Ball then
                    return
                end

                if (
                    Ball.Position -
                    RootPart.Position
                ).Magnitude >
                    Dive.Range then

                    return
                end

                if not IsBallComingToGoalkeeper(
                    Ball
                ) then
                    return
                end

                ExecuteAutoDive(Ball)
            end
        )
end

local function StopAutoDive()
    if AutoDiveConnection then
        AutoDiveConnection:Disconnect()
        AutoDiveConnection = nil
    end
end

local CATCH_REMOTE_NAME = "CatchBall"
local NEAREST_BALL_TTL = 0.05

local AutoCatch = {
    Enabled = false,
    Range = 50,
    Cooldown = 0.25,
    LastCatch = 0,
    MobileRefs = nil,
    _lastPrune = 0
}

function fireCatchBall(ball)
    if not ball
        or not ball.Parent then

        return false
    end

    local remote =
        ReplicatedStorage:FindFirstChild(
            CATCH_REMOTE_NAME
        )

    if remote
        and remote:IsA("RemoteEvent") then

        AutoCatch.LastCatch =
            clock()

        pcall(function()
            remote:FireServer(ball)
        end)

        return true
    end

    return false
end

scanBalls()

do
    local cache = {
        ball = nil,
        dist = math.huge,
        stamp = 0
    }

    function getNearestBall()
        local now =
            clock()

        if now - cache.stamp <
            NEAREST_BALL_TTL then

            return
                cache.ball,
                cache.dist
        end

        cache.stamp =
            now

        local hrp =
            State.HRP

        if not hrp then
            cache.ball = nil
            cache.dist =
                math.huge

            return
                nil,
                math.huge
        end

        local hrpPos =
            hrp.Position

        local best = nil
        local bestDistSq =
            math.huge

        for i = 1, #ballList do
            local ball =
                ballList[i]

            if ball
                and ball.Parent then

                local delta =
                    hrpPos -
                    ball.Position

                local d2 =
                    delta:Dot(delta)

                if d2 < bestDistSq then
                    bestDistSq =
                        d2

                    best = ball
                end
            end
        end

        cache.ball =
            best

        cache.dist =
            best
            and math.sqrt(bestDistSq)
            or math.huge

        return
            best,
            cache.dist
    end
end

        or Humanoid.SeatPart then

        Humanoid:GetState()

        Enum.HumanoidStateType.Running
        and StateValue ~=
        Enum.HumanoidStateType.RunningNoPhysics
        and StateValue ~=
        Enum.HumanoidStateType.Jumping
        and StateValue ~=
        Enum.HumanoidStateType.Landed
        and StateValue ~=
        Enum.HumanoidStateType.Freefall

local function TouchBall(Ball)

        or not Ball.Parent
        or not RootPart then

        return
    end

    pcall(function()
        firetouchinterest(
            RootPart,

            0

        firetouchinterest(
            RootPart,
            Ball,
            1
        )
    end)

        return
    end

        Character:FindFirstChild(
            "Left Foot"
        )
        or Character:FindFirstChild(
            "Left Leg"
        )

    local RightFoot =
        Character:FindFirstChild(
            "Right Foot"

        or Character:FindFirstChild(
            "Right Leg"
        )

        pcall(function()
            firetouchinterest(
                LeftFoot,
                Ball,
                0
            )

            firetouchinterest(
                LeftFoot,
                Ball,
                1
            )
        end)
    end

    if RightFoot then
        pcall(function()
            firetouchinterest(
                RightFoot,
                Ball,
                0
            )

            firetouchinterest(
                RightFoot,
                Ball,
                1
            )
        end)
    end
end

local function ApplySilentAim()
    if not SilentAim
        or not RootPart then

        return
    end

    local Ball =
        GetBall()

    if not Ball then
        return
    end

    local Direction =
        Ball.Position -
        RootPart.Position

    local Flat =
        Vector3.new(
            Direction.X,
            0,
            Direction.Z
        )

    if Flat.Magnitude <= 0 then
        return
    end

    Flat =
        Flat.Unit

    local Look =
        RootPart.CFrame.LookVector

    local FlatLook =
        Vector3.new(
            Look.X,
            0,
            Look.Z
        )

    if FlatLook.Magnitude <= 0 then
        return
    end

    FlatLook =
        FlatLook.Unit

    local Angle =
        math.deg(
            math.acos(
                math.clamp(
                    FlatLook:Dot(Flat),
                    -1,
                    1
                )
            )
        )

    if Angle > MaxAngle then
        return
    end

    local NewDirection =
        FlatLook:Lerp(
            Flat,
            math.clamp(
                AimStrength,
                0,
                1
            )
        )

    RootPart.CFrame =
        CFrame.lookAt(
            RootPart.Position,
            RootPart.Position +
            NewDirection
        )
end

local function ApplyPowerShoot()
    if not PowerShoot then
        return
    end

    local Ball =
        GetBall()

    if not Ball
        or not RootPart then

        return
    end

    if (
        Ball.Position -
        RootPart.Position
    ).Magnitude >
        PowerShootRange then

        return
    end

    Ball.AssemblyLinearVelocity =
        Camera.CFrame.LookVector.Unit
        * ShootPower
end

local function ApplyCurveOnLong()
    if not MoreCurve then
        return
    end

    local Ball =
        GetBall()

    if not Ball then
        return
    end

    local Velocity =
        Ball.AssemblyLinearVelocity

    if Velocity.Magnitude <= 0 then
        return
    end

    local Forward =
        Velocity.Unit

    local Right =
        Camera.CFrame.RightVector

    Ball.AssemblyLinearVelocity =
        Forward *
        Velocity.Magnitude
        +
        Right *
        CurvePower
end

RunService.Heartbeat:Connect(function()
    local now =
        clock()

    if not State.Character
        or not State.Character.Parent then

        UpdateCharacter()
    end

    if AutoCatch.Enabled then
        local hum =
            State.Humanoid

        local hrp =
            State.HRP

        if hum
            and hrp
            and hum.Health > 0
            and now -
                AutoCatch.LastCatch >=
                AutoCatch.Cooldown then

            local ball, dist =
                getNearestBall()

            if ball
                and dist <=
                    AutoCatch.Range then

                fireCatchBall(ball)
            end
        end
    end

    UpdateACHitbox()

    if ESPBall then
        UpdateESP()
    end
end)

task.spawn(function()
    while task.wait(0.05) do
        scanBalls()
    end
end)

RunService.PreRender:Connect(function()
    if not AutoFollowEnabled then
        return
    end

    if not RootPart
        or not RootPart.Parent then

        return
    end

    local Ball =
        GetBall()

    if not Ball then
        return
    end

    if not HabilidadeAtiva()
        and not IsManualInputActive() then

        local Delta =
            Ball.Position -
            RootPart.Position

        local MoveDirection =
            Vector3.new(
                Delta.X,
                0,
                Delta.Z
            )

        if MoveDirection.Magnitude > 0 then
            Humanoid:Move(
                MoveDirection.Unit,
                false
            )
        end
    end
end)

RunService.Heartbeat:Connect(function()
    if ReachEnabled then
        local Ball =
            GetBall()

        if Ball
            and RootPart
            and not HabilidadeAtiva() then

            if (
                Ball.Position -
                RootPart.Position
            ).Magnitude <=
                ReachDistance then

                TouchBall(Ball)
            end
        end
    end

    if PowerShoot then
        ApplyPowerShoot()
    end

    if MoreCurve then
        ApplyCurveOnLong()
    end
end)
