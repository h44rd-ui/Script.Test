local GKBotoes = {}

local function FindTPS()
    local Folder = Workspace:FindFirstChild("WorkspaceLeaderboards")

    if Folder then
        local TPS = Folder:FindFirstChild("TPS")

        if TPS then
            return TPS
        end
    end

        if Object.Name == "TPS" then
            return Object
        end
    end

    return nil
end

local function GetValidBall(Object)
    if not Object then
        return nil
    end

    if Object:IsA("BasePart") then
        return Object
    end

    if Object:IsA("Model") then
        if Object.PrimaryPart then
            return Object.PrimaryPart
        end

        return Object:FindFirstChildWhichIsA("BasePart", true)
    end

    return nil
end

local function GetBall()
    return GetValidBall(FindTPS())
end

local function EscanearBotoesGK()
    GKBotoes = {}

    local PlayerGui = Player:FindFirstChildOfClass("PlayerGui")

    if not PlayerGui then
        return
    end

    for _, Object in ipairs(PlayerGui:GetDescendants()) do
        if Object:IsA("TextButton") or Object:IsA("ImageButton") then
            local Name = string.lower(Object.Name)
            local Text = ""

            pcall(function()
                Text = string.lower(Object.Text or "")
            end)

            if Name:find("gk")
                or Name:find("dive")
                or Name:find("catch")
                or Name:find("high")
                or Name:find("low")
                or Name:find("reflex")
                or Name:find("forward")
                or Name:find("front")
                or Name:find("rush")
                or Text:find("gk")
                or Text:find("dive")
                or Text:find("catch")
                or Text:find("high")
                or Text:find("low")
                or Text:find("reflex")
                or Text:find("forward")
                or Text:find("front")
                or Text:find("rush") then

                table.insert(GKBotoes, Object)
            end
        end
    end
end

local function EncontrarBotaoPorTexto(Texto)
    local Procurado = string.lower(Texto)

    for _, Button in ipairs(GKBotoes) do
        if Button and Button.Parent then
            local Name = string.lower(Button.Name)
            local Text = ""

            pcall(function()
                Text = string.lower(Button.Text or "")
            end)

            if Name == Procurado or Text == Procurado then
                return Button
            end
        end
    end

    for _, Button in ipairs(GKBotoes) do
        if Button and Button.Parent then
            local Name = string.lower(Button.Name)
            local Text = ""

            pcall(function()
                Text = string.lower(Button.Text or "")
            end)

            if Name:find(Procurado, 1, true)
                or Text:find(Procurado, 1, true) then

                return Button
            end
        end
    end

    return nil
end

local function ClicarBotao(Button)
    if not Button or typeof(firesignal) ~= "function" then
        return false
    end

    local Signals = {
        Button.Activated,
        Button.MouseButton1Click,
        Button.MouseButton1Down,
        Button.MouseButton1Up,
        Button.TouchTap
    }

    for _, Signal in ipairs(Signals) do
        pcall(function()
            firesignal(Signal)
        end)
    end

    return true
end

local function GetDivePrediction(Ball)
    if not Ball or not RootPart then
        return nil
    end

    local Position = Ball.Position
    local Velocity = Ball.AssemblyLinearVelocity
    local Speed = Velocity.Magnitude

    if Speed < Dive.MinBallSpeed then
        return nil
    end

    local Distance = (Position - RootPart.Position).Magnitude

    local TravelTime =
        math.clamp(
            Distance / Speed,
            0,
            Dive.Prediction
        )

    local Predicted =
        Position + Velocity * TravelTime

    local Relative =
        Predicted - RootPart.Position

    local Horizontal =
        Vector3.new(
            Relative.X,
            0,
            Relative.Z
        )

    local Right =
        RootPart.CFrame.RightVector

    local RightHorizontal =
        Vector3.new(
            Right.X,
            0,
            Right.Z
        )

    local Side =
        Horizontal:Dot(RightHorizontal)

    return {
        Position = Predicted,
        Side = Side,
        Height = Relative.Y,
        IsCentral = math.abs(Side) <= Dive.CenterWidth,
        IsHigh = Relative.Y >= Dive.HighHeight,
        IsFront = Relative.Y <= Dive.FrontHeight
    }
end

local function GetDiveAction(Data)
    if not Data then
        return nil
    end

    if Data.IsHigh and Data.IsCentral then
        return DiveButtons.HighCatch
    end

    if Data.IsFront and Data.IsCentral then
        return DiveButtons.Front
    end

    if Data.IsHigh then
        return Data.Side < 0
            and DiveButtons.HighLeft
            or DiveButtons.HighRight
    end

    return Data.Side < 0
        and DiveButtons.LowLeft
        or DiveButtons.LowRight
end

local function ExecuteDiveAction(Action)
    if not Action then
        return false
    end

    local Button =
        EncontrarBotaoPorTexto(Action)

    if not Button then
        EscanearBotoesGK()
        Button =
            EncontrarBotaoPorTexto(Action)
    end

    if not Button then
        return false
    end

    if Action == DiveButtons.HighCatch
        and Humanoid then

        pcall(function()
            Humanoid:ChangeState(
                Enum.HumanoidStateType.Jumping
            )
        end)

        task.wait(0.05)
    end

    return ClicarBotao(Button)
end

local function IsBallComingToGoalkeeper(Ball)
    if not Ball or not RootPart then
        return false
    end

    local Velocity =
        Vector3.new(
            Ball.AssemblyLinearVelocity.X,
            0,
            Ball.AssemblyLinearVelocity.Z
        )

    if Velocity.Magnitude < Dive.MinBallSpeed then
        return false
    end

    local Direction =
        RootPart.Position - Ball.Position

    Direction =
        Vector3.new(
            Direction.X,
            0,
            Direction.Z
        )

    if Direction.Magnitude <= 0 then
        return false
    end

    local Dot =
        Velocity.Unit:Dot(Direction.Unit)

    return Dot >= Dive.MinApproachDot
end

local function ExecuteAutoDive(Ball)
    local Data =
        GetDivePrediction(Ball)

    if not Data then
        return
    end

    local Action =
        GetDiveAction(Data)

    if ExecuteDiveAction(Action) then
        Dive.LastAction = tick()
    end
end

local AutoDiveConnection

local function StartAutoDive()
    if AutoDiveConnection then
        return
    end

    AutoDiveConnection =
        RunService.Heartbeat:Connect(function()

            if not Dive.Enabled then
                return
            end

            if not RootPart
                or not RootPart.Parent then

                UpdateCharacter()
                return
            end

            if tick() - Dive.LastAction
                < Dive.Cooldown then

                return
            end

            local Ball =
                GetValidBall(FindTPS())

            if not Ball then
                return
            end

            if (Ball.Position - RootPart.Position).Magnitude
                > Dive.Range then

                return
            end

            if not IsBallComingToGoalkeeper(Ball) then
                return
            end

            ExecuteAutoDive(Ball)
        end)
end

local function StopAutoDive()
    if AutoDiveConnection then
        AutoDiveConnection:Disconnect()
        AutoDiveConnection = nil
    end
end

local function ApplySilentAim()
    if not SilentAim
        or not RootPart
        or not RootPart.Parent then

        return
    end

    local Ball = GetBall()

    if not Ball then
        return
    end

    local Direction =
        Ball.Position - RootPart.Position

    local FlatDirection =
        Vector3.new(
            Direction.X,
            0,
            Direction.Z
        )

    if FlatDirection.Magnitude <= 0 then
        return
    end

    FlatDirection =
        FlatDirection.Unit

    local Look =
        RootPart.CFrame.LookVector

    local FlatLook =
        Vector3.new(
            Look.X,
            0,
            Look.Z
        )

    if FlatLook.Magnitude <= 0 then
        return
    end

    FlatLook =
        FlatLook.Unit

    local Angle =
        math.deg(
            math.acos(
                math.clamp(
                    FlatLook:Dot(FlatDirection),
                    -1,
                    1
                )
            )
        )

    if Angle > MaxAngle then
        return
    end

    local NewDirection =
        FlatLook:Lerp(
            FlatDirection,
            math.clamp(
                AimStrength,
                0,
                1
            )
        )

    RootPart.CFrame =
        CFrame.lookAt(
            RootPart.Position,
            RootPart.Position + NewDirection
        )
end

local ToolConnections = {}

local function ConnectTool(Name, Callback)
    if ToolConnections[Name] then
        ToolConnections[Name]:Disconnect()
    end

    local Tool =
        GetTool(Name)

    if not Tool then
        return
    end

    ToolConnections[Name] =
        Tool.Activated:Connect(function()
            task.defer(Callback)
        end)
end

local function ApplyPowerOnShoot()
    if not PowerShoot then
        return
    end

    local Ball = GetBall()

    if not Ball or not RootPart then
        return
    end

    if (Ball.Position - RootPart.Position).Magnitude
        > PowerShootRange then

        return
    end

    Ball.AssemblyLinearVelocity =
        Camera.CFrame.LookVector.Unit
        * ShootPower
end

local function ApplyCurveOnLong()
    if not MoreCurve then
        return
    end

    local Ball = GetBall()

    if not Ball then
        return
    end

    local Velocity =
        Ball.AssemblyLinearVelocity

    if Velocity.Magnitude <= 0 then
        return
    end

    local Forward =
        Velocity.Unit

    local Right =
        Camera.CFrame.RightVector

    Ball.AssemblyLinearVelocity =
        Forward * Velocity.Magnitude
        + Right * CurvePower
end

local function ConnectShootTools()
    ConnectTool(
        "Shoot",
        ApplyPowerOnShoot
    )

    ConnectTool(
        "Long",
        ApplyCurveOnLong
    )
end

local CATCH_REMOTE_NAME = "CatchBall"

local AutoCatch = {
    Enabled = false,
    Range = 50,
    Cooldown = 0.25,
    LastCatch = 0
}

local NEAREST_BALL_TTL = 0.05

local ballList = {}

local cache = {
    ball = nil,
    dist = math.huge,
    stamp = 0
}

local function RefreshBallList()
    table.clear(ballList)

    local Folder =
        Workspace:FindFirstChild(
            "WorkspaceLeaderboards"
        )

    if Folder then
        local TPS =
            Folder:FindFirstChild("TPS")

        if TPS then
            local Ball =
                GetValidBall(TPS)

            if Ball then
                ballList[#ballList + 1] =
                    Ball
            end
        end
    end

    for _, Object in ipairs(
        Workspace:GetDescendants()
    ) do

        if Object.Name == "TPS" then
            local Ball =
                GetValidBall(Object)

            if Ball then
                local Exists = false

                for i = 1, #ballList do
                    if ballList[i] == Ball then
                        Exists = true
                        break
                    end
                end

                if not Exists then
                    ballList[#ballList + 1] =
                        Ball
                end
            end
        end
    end
end

local function getNearestBall()
    local now = clock()

    if now - cache.stamp
        < NEAREST_BALL_TTL then

        return cache.ball, cache.dist
    end

    cache.stamp = now

    local hrp = RootPart

    if not hrp then
        cache.ball = nil
        cache.dist = math.huge

        return nil, math.huge
    end

    local hrpPos =
        hrp.Position

    local best = nil
    local bestDistSq = math.huge

    for i = 1, #ballList do
        local ball =
            ballList[i]

        if ball and ball.Parent then
            local delta =
                hrpPos - ball.Position

            local d2 =
                delta:Dot(delta)

            if d2 < bestDistSq then
                bestDistSq = d2
                best = ball
            end
        end
    end

    cache.ball = best

    cache.dist =
        best
        and math.sqrt(bestDistSq)
        or math.huge

    return cache.ball, cache.dist
end

local function fireCatchBall(ball)
    if not ball or not ball.Parent then
        return false
    end

    local remote =
        ReplicatedStorage:FindFirstChild(
            CATCH_REMOTE_NAME
        )

    if not remote
        or not remote:IsA("RemoteEvent") then

        return false
    end

    AutoCatch.LastCatch = clock()

    pcall(function()
        remote:FireServer(ball)
    end)

    return true
end

local function HabilidadeAtiva()
    if not Humanoid then
        return true

    if Humanoid.Sit or Humanoid.SeatPart then
        return true

    local State =
        Humanoid:GetState()

    return State ~= Enum.HumanoidStateType.Running
        and State ~= Enum.HumanoidStateType.RunningNoPhysics
        and State ~= Enum.HumanoidStateType.Jumping
        and State ~= Enum.HumanoidStateType.Landed
        and State ~= Enum.HumanoidStateType.Freefall
end

local function TouchBall(Ball)
    if not Ball
        or not Ball.Parent
        or not RootPart then

        return
    end

    pcall(function()
        firetouchinterest(
            RootPart,
            Ball,
            0
        )

        firetouchinterest(
            RootPart,
            Ball,
            1
        )
    end)

    local LeftFoot =
        Character and (
            Character:FindFirstChild(
                "Left Foot"
            )
            or Character:FindFirstChild(
                "Left Leg"
            )
        )

    local RightFoot =
        Character and (
            Character:FindFirstChild(
                "Right Foot"
            )
            or Character:FindFirstChild(
                "Right Leg"
            )
        )

    if LeftFoot then
        pcall(function()
            firetouchinterest(
                LeftFoot,
                Ball,
                0
            )

            firetouchinterest(
                LeftFoot,
                Ball,
                1
            )
        end)
    end

    if RightFoot then
        pcall(function()
            firetouchinterest(
                RightFoot,
                Ball,
                0
            )

            firetouchinterest(
                RightFoot,
                Ball,
                1
            )
        end)
    end
end

local CurveBallConnection

local function AplicarCurva(
    Ball,
    DeltaTime
)

    if not Ball or not Ball.Parent then
        return
    end

    local Velocity =
        Ball.AssemblyLinearVelocity

    local Horizontal =
        Vector3.new(
            Velocity.X,
            0,
            Velocity.Z
        )

    if Horizontal.Magnitude < 5 then
        return
    end

    local Forward =
        Horizontal.Unit

    local Force

    if CurveBallMode == "Direita" then
        Force =
            Vector3.new(
                Forward.Z,
                0,
                -Forward.X
            )

    elseif CurveBallMode == "Esquerda" then
        Force =
            Vector3.new(
                -Forward.Z,
                0,
                Forward.X
            )

    elseif CurveBallMode == "Cima" then
        Force =
            Vector3.new(
                0,
                1,
                0
            )
    else
        return
    end

    local CurveForce =
        Force
        * CurveBallForce
        * DeltaTime
        * 60

    Ball.AssemblyLinearVelocity =
        Velocity + CurveForce
end

local function StartCurveBall()
    if CurveBallConnection then
        return
    end

    CurveBallConnection =
        RunService.Heartbeat:Connect(
            function(DeltaTime)

                if CurveBallMode ==
                    "Desativar" then

                    return
                end

                if CurveBallOnKick
                    and not CurveBallAlwaysActive
                    and tick() >
                        CurveBallKickEnd then

                    return
                end

                local Ball =
                    GetBall()

                if Ball then
                    AplicarCurva(
                        Ball,
                        DeltaTime
                    )
                end
            end
        )
end

local function StopCurveBall()
    if CurveBallConnection then
        CurveBallConnection:Disconnect()
        CurveBallConnection = nil
    end
end

local CurveGroupBall =
    Tabs.Ball:AddRightGroupbox(
        "Curve Ball"
    )

CurveGroupBall:AddDropdown(
    "CurveMode",
    {
        Values = {
            "Desativar",
            "Direita",
            "Esquerda",
            "Cima"
        },

        Default = "Desativar",
        Multi = false,
        Text = "Direction",

        Callback = function(Value)
            CurveBallMode = Value

            if Value ==
                "Desativar" then

                StopCurveBall()
            else
                StartCurveBall()
            end
        end
    }
)

        Max = 100,

        Callback = function(Value)

local LastPowerCheck = 0

RunService.Heartbeat:Connect(function()

    local Now = tick()

    RefreshBallList()

    if SilentAim then
        ApplySilentAim()
    end

    if ReachEnabled
        and Now - LastReachCheck >=
            ReachInterval then

        LastReachCheck = Now

        local Ball = GetBall()

        if Ball
            and RootPart
            and not HabilidadeAtiva() then

            if (Ball.Position -
                RootPart.Position).Magnitude
                <= ReachDistance then

                TouchBall(Ball)
            end
        end
    end

    if PowerShoot
        and Now - LastPowerCheck >=
            0.08 then

        LastPowerCheck = Now

        local Ball = GetBall()

        if Ball and RootPart then

            local Distance =
                (Ball.Position -
                RootPart.Position).Magnitude

            if Distance <=
                PowerShootRange then

                Ball.AssemblyLinearVelocity =
                    Camera.CFrame.LookVector.Unit
                    * ShootPower
            end
        end
    end

    if AutoCatch.Enabled then

        local CatchNow = clock()

        local Hum = Humanoid
        local HRP = RootPart

        if Hum
            and HRP
            and Hum.Health > 0
            and CatchNow -
                AutoCatch.LastCatch >=
                AutoCatch.Cooldown then

            local Ball, Distance =
                getNearestBall()

            if Ball
                and Distance <=
                    AutoCatch.Range then

                fireCatchBall(Ball)
            end
        end
    end

    if ESPBall then
        UpdateESP()
    end
end)

RunService.PreRender:Connect(function()

    if not AutoFollowEnabled then
        return
    end

    if not RootPart
        or not RootPart.Parent then

        return
    end

    local Ball = GetBall()

    if not Ball then
        return
    end

    if not HabilidadeAtiva()
        and not IsManualInputActive() then

        local Delta =
            Ball.Position -
            RootPart.Position

        local MoveDirection =
            Vector3.new(
                Delta.X,
                0,
                Delta.Z
            )

        if MoveDirection.Magnitude > 0 then
            Humanoid:Move(
                MoveDirection.Unit,
                false
            )
        end
    end
end)
