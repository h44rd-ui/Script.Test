local GKBotoes = {}

local function FindTPS()
    local Folder = Workspace:FindFirstChild("WorkspaceLeaderboards")

    if Folder then
        local TPS = Folder:FindFirstChild("TPS")

        if TPS then
            return TPS
        end
    end

    for _, Object in ipairs(Workspace:GetDescendants()) do
        if Object.Name == "TPS" then
            return Object
        end
    end

    return nil
end

local function GetValidBall(Object)
    Object = Object or FindTPS()

    if not Object then
        return nil
    end

    if Object:IsA("BasePart") then
        return Object
    end

    if Object:IsA("Model") then
        if Object.PrimaryPart then
            return Object.PrimaryPart
        end

        return Object:FindFirstChildWhichIsA("BasePart", true)
    end

    return nil
end

local function PegarTodasBolas()
    local Balls = {}

    for _, Object in ipairs(Workspace:GetDescendants()) do
        if Object:IsA("BasePart") or Object:IsA("MeshPart") then
            local Name = Object.Name:lower()

            if Name == "tps" or Name == "ball" or Name == "bola" then
                local Size = Object.Size

                if Size.X >= 0.5 and Size.X <= 8 then
                    table.insert(Balls, Object)
                end
            end
        end
    end

    return Balls
end

local function EscanearBotoesGK()
    GKBotoes = {}

    local PlayerGui = Player:FindFirstChildOfClass("PlayerGui")

    if not PlayerGui then
        return
    end

    for _, Object in ipairs(PlayerGui:GetDescendants()) do
        if Object:IsA("TextButton") or Object:IsA("ImageButton") then
            local Name = string.lower(Object.Name)
            local Text = ""

            pcall(function()
                Text = string.lower(Object.Text or "")
            end)

            if Name:find("gk")
                or Name:find("dive")
                or Name:find("catch")
                or Name:find("high")
                or Name:find("low")
                or Name:find("reflex")
                or Name:find("forward")
                or Name:find("front")
                or Name:find("rush")
                or Text:find("gk")
                or Text:find("dive")
                or Text:find("catch")
                or Text:find("high")
                or Text:find("low")
                or Text:find("reflex")
                or Text:find("forward")
                or Text:find("front")
                or Text:find("rush") then

                table.insert(GKBotoes, Object)
            end
        end
    end
end

local function EncontrarBotaoPorTexto(Texto)
    local Procurado = string.lower(Texto)

    for _, Button in ipairs(GKBotoes) do
        if Button and Button.Parent then
            local Name = string.lower(Button.Name)
            local Text = ""

            pcall(function()
                Text = string.lower(Button.Text or "")
            end)

            if Name == Procurado or Text == Procurado then
                return Button
            end
        end
    end

    for _, Button in ipairs(GKBotoes) do
        if Button and Button.Parent then
            local Name = string.lower(Button.Name)
            local Text = ""

            pcall(function()
                Text = string.lower(Button.Text or "")
            end)

            if Name:find(Procurado, 1, true)
                or Text:find(Procurado, 1, true) then
                return Button
            end
        end
    end

    return nil
end

local function ClicarBotao(Button)
    if not Button or typeof(firesignal) ~= "function" then
        return false
    end

    local Signals = {
        Button.Activated,
        Button.MouseButton1Click,
        Button.MouseButton1Down,
        Button.MouseButton1Up,
        Button.TouchTap
    }

    for _, Signal in ipairs(Signals) do
        pcall(function()
            firesignal(Signal)
        end)
    end

    return true
end

local function GetDivePrediction(Ball)
    if not Ball or not RootPart then
        return nil
    end

    local Position = Ball.Position
    local Velocity = Ball.AssemblyLinearVelocity
    local Speed = Velocity.Magnitude

    if Speed < Dive.MinBallSpeed then
        return nil
    end

    local Distance = (Position - RootPart.Position).Magnitude
    local TravelTime = math.clamp(Distance / Speed, 0, Dive.Prediction)

    local Predicted = Position + Velocity * TravelTime
    local Relative = Predicted - RootPart.Position
    local Horizontal = Vector3.new(Relative.X, 0, Relative.Z)

    local Right = RootPart.CFrame.RightVector
    local RightHorizontal = Vector3.new(Right.X, 0, Right.Z)

    local Side = Horizontal:Dot(RightHorizontal)

    return {
        Position = Predicted,
        Side = Side,
        Height = Relative.Y,
        IsCentral = math.abs(Side) <= Dive.CenterWidth,
        IsHigh = Relative.Y >= Dive.HighHeight,
        IsFront = Relative.Y <= Dive.FrontHeight
    }
end

local function GetDiveAction(Data)
    if not Data then
        return nil
    end

    if Data.IsHigh and Data.IsCentral then
        return DiveButtons.HighCatch
    end

    if Data.IsFront and Data.IsCentral then
        return DiveButtons.Front
    end

    if Data.IsHigh then
        return Data.Side < 0
            and DiveButtons.HighLeft
            or DiveButtons.HighRight
    end

    return Data.Side < 0
        and DiveButtons.LowLeft
        or DiveButtons.LowRight
end

local function ExecuteDiveAction(Action)
    if not Action then
        return false
    end

    local Button = EncontrarBotaoPorTexto(Action)

    if not Button then
        EscanearBotoesGK()
        Button = EncontrarBotaoPorTexto(Action)
    end

    if not Button then
        return false
    end

    if Action == DiveButtons.HighCatch and Humanoid then
        pcall(function()
            Humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
        end)

        task.wait(0.05)
    end

    return ClicarBotao(Button)
end

local function IsBallComingToGoalkeeper(Ball)
    if not Ball or not RootPart then
        return false
    end

    local Velocity = Vector3.new(
        Ball.AssemblyLinearVelocity.X,
        0,
        Ball.AssemblyLinearVelocity.Z
    )

    if Velocity.Magnitude < Dive.MinBallSpeed then
        return false
    end

    local Direction = RootPart.Position - Ball.Position
    Direction = Vector3.new(Direction.X, 0, Direction.Z)

    if Direction.Magnitude <= 0 then
        return false
    end

    local Dot = Velocity.Unit:Dot(Direction.Unit)

    return Dot >= Dive.MinApproachDot
end

local function ExecuteAutoDive(Ball)
    local Data = GetDivePrediction(Ball)

    if not Data then
        return
    end

    local Action = GetDiveAction(Data)

    if ExecuteDiveAction(Action) then
        Dive.LastAction = tick()
    end
end

local AutoDiveConnection

local function StartAutoDive()
    if AutoDiveConnection then
        return
    end

    AutoDiveConnection = RunService.Heartbeat:Connect(function()
        if not Dive.Enabled then
            return
        end

        if not RootPart or not RootPart.Parent then
            UpdateCharacter()
            return
        end

        if tick() - Dive.LastAction < Dive.Cooldown then
            return
        end

        local Ball = GetValidBall()

        if not Ball then
            return
        end

        if (Ball.Position - RootPart.Position).Magnitude > Dive.Range then
            return
        end

        if not IsBallComingToGoalkeeper(Ball) then
            return
        end

        ExecuteAutoDive(Ball)
    end)
end

local function StopAutoDive()
    if AutoDiveConnection then
        AutoDiveConnection:Disconnect()
        AutoDiveConnection = nil
    end
end

local function ApplySilentAim()
    if not SilentAim or not RootPart or not RootPart.Parent then
        return
    end

    local Ball = GetValidBall()

    if not Ball then
        return
    end

    local Direction = Ball.Position - RootPart.Position
    local FlatDirection = Vector3.new(Direction.X, 0, Direction.Z)

    if FlatDirection.Magnitude <= 0 then
        return
    end

    FlatDirection = FlatDirection.Unit

    local Look = RootPart.CFrame.LookVector
    local FlatLook = Vector3.new(Look.X, 0, Look.Z)

    if FlatLook.Magnitude <= 0 then
        return
    end

    FlatLook = FlatLook.Unit

    local Angle = math.deg(math.acos(
        math.clamp(FlatLook:Dot(FlatDirection), -1, 1)
    ))

    if Angle > MaxAngle then
        return
    end

    local NewDirection = FlatLook:Lerp(
        FlatDirection,
        math.clamp(AimStrength, 0, 1)
    )

    RootPart.CFrame = CFrame.lookAt(
        RootPart.Position,
        RootPart.Position + NewDirection
    )
end

local function HabilidadeAtiva()
    if not Humanoid then
        return true
    end

    if Humanoid.Sit or Humanoid.SeatPart then
        return true
    end

    local State = Humanoid:GetState()

    if State ~= Enum.HumanoidStateType.Running
        and State ~= Enum.HumanoidStateType.RunningNoPhysics
        and State ~= Enum.HumanoidStateType.Jumping
        and State ~= Enum.HumanoidStateType.Landed
        and State ~= Enum.HumanoidStateType.Freefall then
        return true
    end

    return false
end

local function IsManualInputActive()
    return UserInputService:IsKeyDown(Enum.KeyCode.W)
        or UserInputService:IsKeyDown(Enum.KeyCode.A)
        or UserInputService:IsKeyDown(Enum.KeyCode.S)
        or UserInputService:IsKeyDown(Enum.KeyCode.D)
end

local function TouchBall(Ball)
    if not Ball or not Ball.Parent then
        return
    end

    if not RootPart or not RootPart.Parent then
        return
    end

    pcall(function()
        firetouchinterest(RootPart, Ball, 0)
        firetouchinterest(RootPart, Ball, 1)
    end)

    local LeftFoot = Character:FindFirstChild("Left Foot")
        or Character:FindFirstChild("Left Leg")

    local RightFoot = Character:FindFirstChild("Right Foot")
        or Character:FindFirstChild("Right Leg")

    if LeftFoot then
        pcall(function()
            firetouchinterest(LeftFoot, Ball, 0)
            firetouchinterest(LeftFoot, Ball, 1)
        end)
    end

    if RightFoot then
        pcall(function()
            firetouchinterest(RightFoot, Ball, 0)
            firetouchinterest(RightFoot, Ball, 1)
        end)
    end
end

local function TeleportToBall()
    local Ball = GetValidBall()

    if not Ball or not Ball.Parent then
        return
    end

    if not RootPart or not RootPart.Parent then
        return
    end

    local Direction = Ball.Position - RootPart.Position

    if Direction.Magnitude <= 0 then
        RootPart.CFrame = CFrame.new(
            Ball.Position + Vector3.new(0, 2, 0)
        )
        return
    end

    local Position = Ball.Position
        - (Direction.Unit * 2.5)
        + Vector3.new(0, 2, 0)

    RootPart.CFrame = CFrame.new(Position)
end

local function FlingBall()
    local Ball = GetValidBall()

    if not Ball or not Ball.Parent then
        return
    end

    if not RootPart or not RootPart.Parent then
        return
    end

    local Direction = Ball.Position - RootPart.Position

    if Direction.Magnitude <= 0 then
        return
    end

    local Position = Ball.Position
        - (Direction.Unit * 1.5)
        + Vector3.new(0, 1, 0)

    RootPart.CFrame = CFrame.new(Position)

    task.wait(0.1)

    if Ball and Ball.Parent then
        local RandomDirection = Vector3.new(
            math.random(-100, 100) / 100,
            1,
            math.random(-100, 100) / 100
        ).Unit

        Ball.AssemblyLinearVelocity =
            RandomDirection * FlingBallForce
            + Vector3.new(0, FlingBallForce * 0.6, 0)

        Ball.AssemblyAngularVelocity = Vector3.new(
            math.random(-50, 50),
            math.random(-50, 50),
            math.random(-50, 50)
        )

        pcall(function()
            firetouchinterest(RootPart, Ball, 0)
            firetouchinterest(RootPart, Ball, 1)
        end)
    end
end

local function CheckPowerShoot()
    if not PowerShootEnabled then
        return
    end

    if not RootPart or not RootPart.Parent then
        return
    end

    local Ball = GetValidBall()

    if not Ball or not Ball.Parent then
        return
    end

    local Distance = (Ball.Position - RootPart.Position).Magnitude

    if Distance > PowerShootRange then
        return
    end

    local Direction = (Ball.Position - RootPart.Position).Unit
    local LookDirection = RootPart.CFrame.LookVector
    local ShootDirection = (LookDirection + Direction).Unit

    Ball.AssemblyLinearVelocity =
        ShootDirection * PowerShootForce
        + Vector3.new(0, PowerShootForce * 0.3, 0)

    Ball.AssemblyAngularVelocity = Vector3.new(
        math.random(-30, 30),
        math.random(-30, 30),
        math.random(-30, 30)
    )

    pcall(function()
        firetouchinterest(RootPart, Ball, 0)
        firetouchinterest(RootPart, Ball, 1)
    end)
end

local function AplicarCurva(Ball, DeltaTime)
    if not Ball or not Ball.Parent then
        return
    end

    local Velocity = Ball.AssemblyLinearVelocity
    local HorizontalVelocity = Vector3.new(
        Velocity.X,
        0,
        Velocity.Z
    )

    if HorizontalVelocity.Magnitude < 5 then
        return
    end

    local Forward = HorizontalVelocity.Unit
    local Force = Vector3.zero

    if CurveBallMode == "Direita" then
        Force = Vector3.new(
            Forward.Z,
            0,
            -Forward.X
        )
    elseif CurveBallMode == "Esquerda" then
        Force = Vector3.new(
            -Forward.Z,
            0,
            Forward.X
        )
    elseif CurveBallMode == "Cima" then
        Force = Vector3.new(0, 1, 0)
    else
        return
    end

    local CurveForce =
        Force * CurveBallForce * DeltaTime * 60

    local FinalVelocity =
        Velocity + CurveForce

    local FinalHorizontal = Vector3.new(
        FinalVelocity.X,
        0,
        FinalVelocity.Z
    )

    if FinalHorizontal.Magnitude <= 0 then
        return
    end

    local ForwardDot =
        FinalHorizontal.Unit:Dot(Forward)

    if ForwardDot < 0.5 then
        local CorrectedHorizontal =
            (Forward * HorizontalVelocity.Magnitude * 0.9)
            + Vector3.new(
                CurveForce.X,
                0,
                CurveForce.Z
            )

        FinalVelocity = Vector3.new(
            CorrectedHorizontal.X,
            FinalVelocity.Y,
            CorrectedHorizontal.Z
        )
    end

    Ball.AssemblyLinearVelocity = FinalVelocity
end

local function StartCurveBall()
    if CurveBallConn then
        return
    end

    CurveBallConn = RunService.Heartbeat:Connect(function(DeltaTime)
        if CurveBallMode == "Desativar" then
            return
        end

        if CurveBallOnKick and not CurveBallAlwaysActive then
            if tick() > CurveBallKickEnd then
                return
            end
        end

        for _, Ball in ipairs(PegarTodasBolas()) do
            AplicarCurva(Ball, DeltaTime)
        end
    end)
end

local function StopCurveBall()
    if CurveBallConn then
        CurveBallConn:Disconnect()
        CurveBallConn = nil
    end
end

local function UpdateCurveBall()
    if CurveBallMode == "Desativar" then
        StopCurveBall()
    elseif not CurveBallConn then
        StartCurveBall()
    end
end

local function DetectarChute()
    local function MarcarChute()
        CurveBallKickEnd =
            tick() + CurveBallKickDuration
    end

    local function HookTool(Tool)
        if not Tool:IsA("Tool") then
            return
        end

        local Name = Tool.Name:lower()

        if Name:find("chutar")
            or Name:find("chute")
            or Name:find("kick")
            or Name:find("shoot")
            or Name:find("bater")
            or Name:find("toque") then

            Tool.Activated:Connect(MarcarChute)
            Tool.Unequipped:Connect(MarcarChute)
        end
    end

    local function HookTools(Container)
        if not Container then
            return
        end

        for _, Object in ipairs(Container:GetChildren()) do
            HookTool(Object)
        end

        Container.ChildAdded:Connect(function(Object)
            task.wait(0.1)
            HookTool(Object)
        end)
    end

    HookTools(Player:FindFirstChild("Backpack"))
    HookTools(Character)

    Player.CharacterAdded:Connect(function(NewCharacter)
        task.wait(1)

        HookTools(Player:FindFirstChild("Backpack"))
        HookTools(NewCharacter)
    end)

    pcall(function()
        local function HookRemote(Object)
            if not Object:IsA("RemoteEvent") then
                return
            end

            local Name = Object.Name:lower()

            if Name:find("chutar")
                or Name:find("chute")
                or Name:find("kick")
                or Name:find("shoot")
                or Name:find("bater") then

                Object.OnClientEvent:Connect(MarcarChute)
            end
        end

        for _, Object in ipairs(ReplicatedStorage:GetDescendants()) do
            HookRemote(Object)
        end

        ReplicatedStorage.DescendantAdded:Connect(function(Object)
            task.wait(0.1)
            HookRemote(Object)
        end)
    end)
end

RunService.PreRender:Connect(function()
    if not RootPart or not RootPart.Parent or not Humanoid then
        return
    end

    local Ball = GetValidBall()
    local Habilidade = HabilidadeAtiva()

    if AutoFollowEnabled
        and not IsManualInputActive()
        and Ball
        and not Habilidade then

        local Delta = Ball.Position - RootPart.Position
        local MoveDirection = Vector3.new(
            Delta.X,
            0,
            Delta.Z
        )

        if MoveDirection.Magnitude > 0.5 then
            Humanoid:Move(
                MoveDirection.Unit,
                false
            )
        end
    end

    if ReachEnabled
        and Ball
        and not Habilidade then

        local Distance =
            (Ball.Position - RootPart.Position).Magnitude

        if Distance <= ReachDistance then
            TouchBall(Ball)
        end
    end
